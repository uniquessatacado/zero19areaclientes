  async function prepareLetteringLayout(item){
    const legacy=await prepareLegacyLetteringLayout(item);
    return refineLetteringLayout(item,legacy,{drawLine:drawPreparedLetteringLine,sanitizeSvg,hasGlyph:(family,char)=>fontCoverageByFamily.get(family)?.hasGlyph(char)===true});
  }

  async function upgradeCurrentLettering(current=()=>true){
    const source=filmItems,generation=productionAccountGeneration,replacements=[],changed=new Set();
    for(const item of source){
      if(item.type!=='team_customization'||item.vectorSource||(!item.name&&!item.number))continue;
      const next={...item,name:String(item.name).normalize('NFC'),letteringMetricsVersion:3,spacingMode:item.spacingMode||'optical'};
      const layout=await prepareLetteringLayout(next);if(!current()||generation!==productionAccountGeneration||filmItems!==source)return null;
      next.widthCm=layout.widthCm;next.heightCm=layout.heightCm;
      const needs=item.letteringMetricsVersion!==3||Math.abs(next.widthCm-item.widthCm)>.000001||Math.abs(next.heightCm-item.heightCm)>.000001;
      if(needs){delete next.previewDataUrl;replacements.push(next);changed.add(item.localId);}
    }
    if(!current()||generation!==productionAccountGeneration||filmItems!==source)return null;
    for(const next of replacements)Object.assign(source.find(i=>i.localId===next.localId),next,{previewDataUrl:undefined});
    if(changed.size)filmMaskCache.clear();
    return changed;
  }
